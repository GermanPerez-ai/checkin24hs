# Comando completo para aplicar serve-crm.js a todos los contenedores

SERVICE_NAME="checkin24hs_crm"
CONTAINER_NAMES=$(docker service ps $SERVICE_NAME --format "{{.Name}}" | sort -u)
for CONTAINER_NAME in $CONTAINER_NAMES; do CONTAINER_ID=$(docker ps -a --filter "name=$CONTAINER_NAME" --format "{{.ID}}" | head -1); if [ ! -z "$CONTAINER_ID" ]; then echo "Copiando a $CONTAINER_NAME ($CONTAINER_ID)"; docker cp /root/checkin24hs/serve-crm.js $CONTAINER_ID:/app/serve-crm.js 2>&1 | head -1; fi; done
docker service update --force $SERVICE_NAME
sleep 40
RUNNING_CONTAINERS=$(docker ps --filter "name=$SERVICE_NAME" --format "{{.ID}} {{.Names}}")
while IFS= read -r line; do CONTAINER_ID=$(echo "$line" | awk '{print $1}'); if [ ! -z "$CONTAINER_ID" ]; then echo "Copiando a nuevo contenedor $CONTAINER_ID"; docker cp /root/checkin24hs/serve-crm.js $CONTAINER_ID:/app/serve-crm.js 2>&1 | head -1; docker exec $CONTAINER_ID ls -lh /app/serve-crm.js 2>&1 | head -1; fi; done <<< "$RUNNING_CONTAINERS"
docker service logs $SERVICE_NAME --tail 30

echo "🔍 Verificando configuración de red del servicio..." && echo "" && echo "1. Verificar que el servicio está en la red easypanel:" && docker service inspect checkin24hs_checkin24hs-dashboard --format '{{range .Spec.TaskTemplate.Networks}}{{.Target}} {{end}}' | xargs -I {} docker network inspect {} --format '{{.Name}}' && echo "" && echo "2. IP actual del contenedor:" && CONTAINER_ID=$(docker ps | grep checkin24hs_checkin24hs-dashboard | awk '{print $1}' | head -1) && CURRENT_IP=$(docker inspect $CONTAINER_ID --format '{{range $key, $value := .NetworkSettings.Networks}}{{if eq $key "easypanel"}}{{$value.IPAddress}}{{end}}{{end}}') && echo "IP: $CURRENT_IP" && echo "" && echo "3. Verificar alias del contenedor:" && docker inspect $CONTAINER_ID --format '{{range $key, $value := .NetworkSettings.Networks}}{{if eq $key "easypanel"}}{{range $value.Aliases}}{{.}} {{end}}{{end}}{{end}}' && echo "" && echo "4. Probar conexión desde Traefik con IP:" && docker exec $(docker ps | grep traefik | awk '{print $1}') wget -O- http://$CURRENT_IP:3000 2>&1 | head -5 && echo "" && echo "5. Verificar logs de Traefik (últimas líneas relacionadas con panel):" && docker logs $(docker ps | grep traefik | awk '{print $1}') --tail 20 | grep -i panel || echo "No se encontraron logs relacionados con panel"

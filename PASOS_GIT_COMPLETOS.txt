# PASOS PARA SUBIR CAMBIOS A GITHUB

# PASO 1: Agregar los archivos al staging
git add Dockerfile
git add serve-dashboard.js
git add dashboard.html
git add deploy/dashboard.html

# PASO 2: Verificar qué se va a commitear
git status

# PASO 3: Hacer el commit
git commit -m "Actualizar Dockerfile y configuración del dashboard"

# PASO 4: Push a GitHub (te pedirá credenciales)
git push origin main
